```powershell
python -m atx_db.stagelake catalog --root <absolute-lake> --out <absolute-catalog.duckdb> --tier1-contract <independent-complete-catalog-contract.json>
python -m atx_db.stagelake verify --root <absolute-lake> --tier1-contract <same-contract.json> --catalog <absolute-catalog.duckdb>
```
